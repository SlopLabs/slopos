# check_quota_headroom gate data — variant: tests
#
#     scripts/check_quota_headroom.sh --variant tests --emit-allowlist
#
# One line per <phase> <TAB> <kind> <TAB> <cap>, where the cap is the
# highest peak any single account row reached for that kind. Measured,
# never chosen: raise a cap only with a fresh --emit-allowlist in the
# same commit, and say in the message what started consuming more.
#
# A cap matching nothing is a dead entry and fails, because a kind that
# stopped being reported looks exactly like a kind that stayed cheap.
#
# The rows that move between runs of one ISO — kernelmeta, pages,
# residentpages, diskblocks, pinnedbytes, task — carry the largest peak
# measured over several captures (a slow AP, or a host without KVM, peaks
# higher); every other row is deterministic and carries its exact peak.

# The peak of a kind nobody exercised is zero, and a cap derived from
# it would bound nothing. Every listed kind must have been reached.
min-kinds 2
min-kinds-for boot 1

# A denial is an over-limit charge. Under quota=warn it is granted and
# counted, so a non-zero total means the enforced tier would have
# refused something — which is a finding, not a pass.
max-denials 0

# The refusals the suite provokes on purpose, held to their exact count:
# fewer means a test stopped refusing, more means something else was.
# Kernel phase: the stest that lowers the root ceiling and over-asks, the
# forked child's write past a full ceiling (tests_commit), and the COW signal
# frame pushed into a full ceiling on the syscall-exit and IRQ-exit roads.
# Userland adds the build-loop over-ask, refused twice because the mmap
# handler reclaims and retries once, and the hog's child's refused write.
expect-denials post-kernel-tests commitpages 4
expect-denials post-userland-tests commitpages 7
# The bomb stops once twenty forks in a row are refused, at a person's task
# share with its children left unreaped and then at their process share.
expect-denials post-userland-tests process 20
expect-denials post-userland-tests task 20

require-phase boot
require-phase post-kernel-tests
require-phase post-userland-tests

# What sets each moving or non-obvious cap:
#   boot kernelmeta            the block layer's, NVMe's, xHCI's and the NIC
#                              table's device state, on a boot with a late AP.
#   kernel commitpages         `test_spawn_delegates_within_the_spawner` holds a
#                              principal beside each child it spawns.
#   kernel kernelmeta          `test_activations_stay_within_the_pool` holds 32
#                              activations at once; the heap keeps those pages.
#   kernel pages               a 1 GiB MAP_NORESERVE reservation held in a
#                              parent and its fork, counted twice.
#   kernel residentpages, task `test_signals_need_the_uid_rule` holds four
#                              principals of 257 resident pages at once.
#   userland commitpages       the build-loop OOM test's hog, which promises
#                              the whole headroom less 3072 pages.
#   userland custody           `privd_test`'s requests, each carrying three or
#                              four descriptors as SCM_RIGHTS.
#   userland diskblocks        `persist_test`'s filler meeting the reserve of
#                              the 128M tests image.
#   userland kernelmeta, process, task, fdslot
#                              `sessions_test`'s fork bomb: a person's 256
#                              processes and 2048 tasks beside the suite's own.

# The charge path's cost, as ratios rather than cycle counts. A cycle
# count here measures the accelerator: the same tree reports ~1500
# cycles per charge under KVM and ~20000 under TCG, and both are
# correct. Every number below is x100, so 420 means 4.20.
#
# The reference floors are emitted at half the observed value, because
# a bare CAS is relatively dearer under TCG than natively and this
# ratio does move with the accelerator. They prove the measurement
# happened; the depth ratio below is the check.
min-charge-over-reference 1 982
min-charge-over-reference 7 3759
max-depth-cost-ratio 7 478

# A physical bound on the reference, not a measurement.
min-reference-cycles 20
